-- =====================================================================
-- NioGabay database schema (Supabase / PostgreSQL)
-- Run ONCE in: Supabase Dashboard -> SQL Editor -> New query -> Run.
--
-- Design notes
--  * One sensor for now, but readings reference a sensors table so more
--    can be added later without redesigning anything.
--  * Water levels are stored in centimetres (what the ultrasonic sensor
--    gives). Alert thresholds are entered in feet (what the barangay
--    uses); alert_levels.level_cm is computed automatically (1 ft = 30.48 cm).
--  * ASSUMED values to confirm with the barangay: orange = 3 ft,
--    red = 4 ft. Change them later with a simple UPDATE on alert_levels.
--  * Residents do NOT log in. They only subscribe to SMS alerts with a
--    phone number (sms_subscribers). Only admin/staff have accounts.
--  * Security: every table has RLS on and no policies, and only the
--    server-side service_role gets access. The browser never talks to
--    these tables directly; the Nuxt server routes do, using the secret key.
-- =====================================================================

begin;

-- ---------- Types ----------
create type public.user_role as enum ('admin', 'staff');

-- ---------- Helper: keep updated_at current ----------
create function public.set_updated_at()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  new.updated_at := now();
  return new;
end;
$$;

-- ---------- Users (admin / staff) ----------
-- Login itself is handled by Supabase Auth (auth.users). This table
-- stores the barangay-specific details and the role.
create table public.profiles (
  id             uuid primary key references auth.users (id) on delete cascade,
  full_name      text not null,
  position_title text,                                  -- e.g. 'Barangay Secretary'
  role           public.user_role not null default 'staff',
  is_active      boolean not null default true,
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now()
);

-- ---------- Sensors ----------
create table public.sensors (
  id                   uuid primary key default gen_random_uuid(),
  name                 text not null,
  location_description text,
  latitude             numeric(9, 6),
  longitude            numeric(9, 6),
  -- Distance from the sensor face down to the ground/riverbed (cm).
  -- water_level = mount_height_cm - measured distance. Set after install.
  mount_height_cm      numeric(6, 1) check (mount_height_cm > 0),
  -- SHA-256 hash of the device API key (never store the key itself).
  api_key_hash         text,
  is_active            boolean not null default true,
  last_seen_at         timestamptz,
  created_at           timestamptz not null default now(),
  updated_at           timestamptz not null default now()
);

-- ---------- Alert levels (thresholds) ----------
create table public.alert_levels (
  id          smallint generated always as identity primary key,
  code        text not null unique,
  label       text not null,
  level_ft    numeric(4, 2) not null unique check (level_ft > 0),
  level_cm    numeric(6, 1) generated always as (round(level_ft * 30.48, 1)) stored,
  description text,
  updated_at  timestamptz not null default now()
);

-- Returns the highest alert level reached by a water level (cm),
-- or 'normal' when it is below the lowest threshold.
create function public.current_alert_level(p_water_level_cm numeric)
returns text
language sql
stable
set search_path = ''
as $$
  select coalesce(
    (select code
       from public.alert_levels
      where level_cm <= p_water_level_cm
      order by level_cm desc
      limit 1),
    'normal'
  );
$$;

-- ---------- Sensor readings ----------
create table public.readings (
  id             bigint generated always as identity primary key,
  sensor_id      uuid not null references public.sensors (id) on delete cascade,
  distance_cm    numeric(6, 1) not null check (distance_cm >= 0),     -- raw sensor value
  water_level_cm numeric(6, 1) not null check (water_level_cm >= 0), -- computed by the server
  recorded_at    timestamptz not null default now()                  -- server receive time
);

create index readings_sensor_time_idx on public.readings (sensor_id, recorded_at desc);

-- ---------- Alerts (automatic from the sensor, or sent manually) ----------
create table public.alerts (
  id              bigint generated always as identity primary key,
  source          text not null check (source in ('sensor', 'manual')),
  alert_level_id  smallint references public.alert_levels (id),
  sensor_id       uuid references public.sensors (id) on delete set null,
  reading_id      bigint references public.readings (id) on delete set null,
  message         text not null,
  created_by      uuid references public.profiles (id) on delete set null, -- null = automatic
  triggered_at    timestamptz not null default now(),
  acknowledged_by uuid references public.profiles (id) on delete set null,
  acknowledged_at timestamptz
);

create index alerts_triggered_idx on public.alerts (triggered_at desc);

-- ---------- SMS subscribers (residents) ----------
create table public.sms_subscribers (
  id           uuid primary key default gen_random_uuid(),
  full_name    text,
  -- Normalize to 09XXXXXXXXX in the app before saving.
  phone_number text not null unique check (phone_number ~ '^09[0-9]{9}$'),
  purok_street text,
  is_active    boolean not null default true,
  consented_at timestamptz not null, -- record when the resident agreed (Data Privacy Act)
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now()
);

-- ---------- SMS delivery log ----------
create table public.sms_logs (
  id                  bigint generated always as identity primary key,
  alert_id            bigint references public.alerts (id) on delete set null,
  subscriber_id       uuid references public.sms_subscribers (id) on delete set null,
  recipient_phone     text not null, -- copy kept even if the subscriber is later deleted
  body                text not null,
  status              text not null default 'queued' check (status in ('queued', 'sent', 'failed')),
  provider_message_id text,
  error_message       text,
  created_at          timestamptz not null default now(),
  sent_at             timestamptz
);

create index sms_logs_alert_idx on public.sms_logs (alert_id);

-- ---------- Announcements ----------
create table public.announcements (
  id           bigint generated always as identity primary key,
  title        text not null,
  body         text not null,
  category     text not null default 'general'
               check (category in ('flood', 'evacuation', 'advisory', 'general')),
  published_at timestamptz,  -- null = draft; set to publish
  created_by   uuid references public.profiles (id) on delete set null,
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now()
);

create index announcements_published_idx on public.announcements (published_at desc);

-- ---------- Evacuation centers ----------
create table public.evacuation_centers (
  id             bigint generated always as identity primary key,
  name           text not null,
  address        text,
  latitude       numeric(9, 6),
  longitude      numeric(9, 6),
  capacity       integer check (capacity >= 0),
  contact_person text,
  contact_number text,
  is_open        boolean not null default false, -- currently open for evacuees
  notes          text,
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now()
);

-- ---------- Emergency contacts ----------
create table public.emergency_contacts (
  id         bigint generated always as identity primary key,
  name       text not null,
  category   text,           -- e.g. 'BDRRMO', 'Fire', 'Police', 'Hospital'
  number     text not null,
  sort_order integer not null default 0,
  is_active  boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- ---------- Standard flood reports ----------
-- Replaces the free-form Messenger reports in Chapter 4, section 4.2
-- (timestamp, flooded areas, roads, evacuees) with one fixed format.
create table public.flood_reports (
  id                    bigint generated always as identity primary key,
  reported_at           timestamptz not null default now(),
  flooded_areas         text,
  passable_roads        text,
  impassable_roads      text,
  families_evacuated    integer not null default 0 check (families_evacuated >= 0),
  individuals_evacuated integer not null default 0 check (individuals_evacuated >= 0),
  remarks               text,
  created_by            uuid references public.profiles (id) on delete set null,
  created_at            timestamptz not null default now()
);

create index flood_reports_reported_idx on public.flood_reports (reported_at desc);

-- ---------- updated_at triggers ----------
create trigger set_updated_at before update on public.profiles
  for each row execute function public.set_updated_at();
create trigger set_updated_at before update on public.sensors
  for each row execute function public.set_updated_at();
create trigger set_updated_at before update on public.alert_levels
  for each row execute function public.set_updated_at();
create trigger set_updated_at before update on public.sms_subscribers
  for each row execute function public.set_updated_at();
create trigger set_updated_at before update on public.announcements
  for each row execute function public.set_updated_at();
create trigger set_updated_at before update on public.evacuation_centers
  for each row execute function public.set_updated_at();
create trigger set_updated_at before update on public.emergency_contacts
  for each row execute function public.set_updated_at();

-- ---------- Starting data ----------
insert into public.alert_levels (code, label, level_ft, description) values
  ('yellow',     'Yellow - Warning',  1.00, 'Water reached 1 ft. Yellow warning on the gauge (Chapter 4, 4.1).'),
  ('evacuation', 'Evacuation',        2.00, 'Barangay starts rescue and evacuation of residents (Chapter 4, 4.1).'),
  ('orange',     'Orange - Alert',    3.00, 'ASSUMED value. Confirm with the barangay.'),
  ('red',        'Red - Critical',    4.00, 'ASSUMED value. Confirm with the barangay.');

insert into public.sensors (name, location_description) values
  ('NioGabay Sensor 1', 'Barangay Niog, Bacoor, Cavite (set exact spot and mount height after installation)');

-- ---------- Security ----------
-- Row Level Security on, with NO policies = anon/authenticated users get nothing.
alter table public.profiles           enable row level security;
alter table public.sensors            enable row level security;
alter table public.alert_levels       enable row level security;
alter table public.readings           enable row level security;
alter table public.alerts             enable row level security;
alter table public.sms_subscribers    enable row level security;
alter table public.sms_logs           enable row level security;
alter table public.announcements      enable row level security;
alter table public.evacuation_centers enable row level security;
alter table public.emergency_contacts enable row level security;
alter table public.flood_reports      enable row level security;

-- Explicitly close the tables to the public Data API roles, whatever the
-- project's "automatically expose new tables" setting was...
revoke all on table
  public.profiles, public.sensors, public.alert_levels, public.readings,
  public.alerts, public.sms_subscribers, public.sms_logs, public.announcements,
  public.evacuation_centers, public.emergency_contacts, public.flood_reports
from anon, authenticated;

-- ...and open them to the server only (Nuxt server routes use the secret key).
grant select, insert, update, delete on table
  public.profiles, public.sensors, public.alert_levels, public.readings,
  public.alerts, public.sms_subscribers, public.sms_logs, public.announcements,
  public.evacuation_centers, public.emergency_contacts, public.flood_reports
to service_role;

revoke execute on function public.set_updated_at() from public, anon, authenticated;
revoke execute on function public.current_alert_level(numeric) from public, anon, authenticated;
grant execute on function public.current_alert_level(numeric) to service_role;

commit;
